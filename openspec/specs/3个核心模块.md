# 模块一：容器编排与生命周期管理 (`orchestrator`)

## 1. 模块定位

负责与底层 Docker Daemon 进行通信，实现 Neko 渲染容器的**动态按需拉起、隔离端口分配、健康等待探测、资源硬限额注入及空闲超时销毁**。

## 2. 核心数据模型 (`session.go` / `types.go`)

```go
package orchestrator

import (
	"sync"
	"time"
)

type SessionStatus string

const (
	StatusPending   SessionStatus = "pending"   // 容器正在拉起并预热
	StatusRunning   SessionStatus = "running"   // Chromium 加载完成，可以推流
	StatusDestroyed SessionStatus = "destroyed" // 已回收
)

type StreamSession struct {
	Token        string        `json:"token"`
	TargetURL    string        `json:"target_url"`
	ContainerID  string        `json:"container_id"`
	ContainerIP  string        `json:"container_ip"`
	HTTPPort     int           `json:"http_port"`      // 容器内 Neko Web 端口
	UDPPortStart int           `json:"udp_port_start"` // WebRTC UDP 范围起始
	UDPPortEnd   int           `json:"udp_port_end"`   // WebRTC UDP 范围结束
	Status       SessionStatus `json:"status"`
	CreatedAt    time.Time     `json:"created_at"`
	LastActiveAt time.Time     `json:"last_active_at"` // 客户端最后心跳时间
}

type SessionManager struct {
	mu           sync.RWMutex
	sessions     map[string]*StreamSession
	portPool     *PortAllocator
	dockerClient *DockerClient
}

```

## 3. 动态端口分配器 (`port_allocator.go`)

每个 Neko 容器需要独占的 WebRTC UDP 端口段（用于 ICE 穿透和推流）。

```go
package orchestrator

import (
	"errors"
	"sync"
)

type PortAllocator struct {
	mu        sync.Mutex
	baseHTTP  int
	usedHTTP  map[int]bool
	baseUDP   int
	stepUDP   int // 单个实例分配的 UDP 端口跨度，如 20 个
	usedUDP   map[int]bool
}

func NewPortAllocator(httpBase, udpBase, step int) *PortAllocator {
	return &PortAllocator{
		baseHTTP: httpBase, // 如 18080
		usedHTTP: make(map[int]bool),
		baseUDP:  udpBase,  // 如 52000
		stepUDP:  step,     // 如 20
		usedUDP:  make(map[int]bool),
	}
}

// 分配一组端口
func (p *PortAllocator) Allocate() (httpPort int, udpStart int, udpEnd int, err error) {
	p.mu.Lock()
	defer p.mu.Unlock()

	// 寻址未使用的 HTTP 端口
	for i := 0; i < 100; i++ {
		candidateHTTP := p.baseHTTP + i
		if !p.usedHTTP[candidateHTTP] {
			httpPort = candidateHTTP
			break
		}
	}
	if httpPort == 0 {
		return 0, 0, 0, errors.New("HTTP 端口池已耗尽")
	}

	// 寻址未使用的 UDP 段
	for i := 0; i < 100; i++ {
		candidateUDPStart := p.baseUDP + (i * p.stepUDP)
		if !p.usedUDP[candidateUDPStart] {
			udpStart = candidateUDPStart
			udpEnd = candidateUDPStart + p.stepUDP - 1
			break
		}
	}
	if udpStart == 0 {
		return 0, 0, 0, errors.New("UDP 端口池已耗尽")
	}

	p.usedHTTP[httpPort] = true
	p.usedUDP[udpStart] = true
	return httpPort, udpStart, udpEnd, nil
}

// 释放归还端口
func (p *PortAllocator) Release(httpPort int, udpStart int) {
	p.mu.Lock()
	defer p.mu.Unlock()
	delete(p.usedHTTP, httpPort)
	delete(p.usedUDP, udpStart)
}

```

## 4. 容器创建与软解参数组装 (`docker_client.go`)

```go
package orchestrator

import (
	"context"
	"fmt"

	"github.com/docker/docker/api/types/container"
	"github.com/docker/docker/api/types/network"
	"github.com/docker/docker/client"
	"github.com/docker/go-connections/nat"
)

type DockerClient struct {
	cli *client.Client
}

func (d *DockerClient) LaunchNekoContainer(ctx context.Context, session *StreamSession, cpuLimit float64, memoryLimitMB int64) error {
	containerName := fmt.Sprintf("gostream3d_%s", session.Token)

	// 1. 环境变量装配：开启 CPU 软解，无感免密与控制权接管
	env := []string{
		"NEKO_PASSWORD_ADMIN=admin_internal_token",
		"NEKO_PASSWORD=user_internal_token",
		"NEKO_IMPLICIT_CONTROL=true", // 访问即自动获取鼠标键盘控制权
		"NEKO_LOCKS=control",
		"NEKO_SCREEN=1920x1080@30",
		fmt.Sprintf("NEKO_WEBRTC_EPR=%d-%d", session.UDPPortStart, session.UDPPortEnd),
		fmt.Sprintf("NEKO_CHROME_ARGS=--kiosk --no-first-run --no-default-browser-check --disable-translate --use-gl=angle --use-angle=swiftshader --ignore-gpu-blocklist --enable-webgl --app=%s", session.TargetURL),
	}

	// 2. 端口绑定声明 (HTTP 与 UDP 范围)
	exposedPorts := nat.PortSet{}
	portBindings := nat.PortMap{}

	// HTTP 端口映射 (8080/tcp -> session.HTTPPort)
	nekoHTTPPort := nat.Port("8080/tcp")
	exposedPorts[nekoHTTPPort] = struct{}{}
	portBindings[nekoHTTPPort] = []nat.PortBinding{
		{HostIP: "0.0.0.0", HostPort: fmt.Sprintf("%d", session.HTTPPort)},
	}

	// UDP 端口映射 (范围映射)
	for p := session.UDPPortStart; p <= session.UDPPortEnd; p++ {
		udpPort := nat.Port(fmt.Sprintf("%d/udp", p))
		exposedPorts[udpPort] = struct{}{}
		portBindings[udpPort] = []nat.PortBinding{
			{HostIP: "0.0.0.0", HostPort: fmt.Sprintf("%d", p)},
		}
	}

	// 3. 宿主机资源限额配置 (针对 8C16G 机器保护)
	resources := container.Resources{
		NanoCPUs: int64(cpuLimit * 1e9),     // 如 2.5 核 = 2500000000
		Memory:   memoryLimitMB * 1024 * 1024, // 如 2048 MB
	}

	// 4. 调用 Docker API
	resp, err := d.cli.ContainerCreate(ctx,
		&container.Config{
			Image:        "m1k1o/neko:chromium",
			Env:          env,
			ExposedPorts: exposedPorts,
		},
		&container.HostConfig{
			PortBindings: portBindings,
			Resources:    resources,
			ShmSize:      1024 * 1024 * 1024, // /dev/shm 设为 1G，防止浏览器渲染崩溃
		},
		&network.NetworkingConfig{},
		nil,
		containerName,
	)
	if err != nil {
		return err
	}

	session.ContainerID = resp.ID
	return d.cli.ContainerStart(ctx, resp.ID, container.StartOptions{})
}

```

## 5. 空闲检测与自动回收守护巡检 (`cleaner.go`)

```go
package orchestrator

import (
	"context"
	"log"
	"time"

	"github.com/docker/docker/api/types/container"
)

// StartIdleCleaner 启动后台协程轮询超时会话
func (sm *SessionManager) StartIdleCleaner(interval time.Duration, idleTimeout time.Duration) {
	go func() {
		ticker := time.NewTicker(interval)
		for range ticker.C {
			sm.mu.Lock()
			now := time.Now()
			for token, session := range sm.sessions {
				// 如果已断开且最后活跃时间超过阈值（如 180 秒）
				if now.Sub(session.LastActiveAt) > idleTimeout {
					log.Printf("[GC] 会话超时无活跃，销毁容器: %s (Token: %s)", session.ContainerID, token)

					// 1. 强制杀死并移除容器
					_ = sm.dockerClient.cli.ContainerRemove(context.Background(), session.ContainerID, container.RemoveOptions{
						Force: true,
					})

					// 2. 归还端口池
					sm.portPool.Release(session.HTTPPort, session.UDPPortStart)

					// 3. 移除内存状态
					delete(sm.sessions, token)
				}
			}
			sm.mu.Unlock()
		}
	}()
}
```

# 模块二：信令透传与 WebRTC 会话代理 (`internal/proxy`)

## 1. 模块定位

负责建立客户端 React 与内部容器 Neko 之间的 **WebSocket 信令桥梁**。对外只暴露 Go 网关的统一信令端口，隐藏内部 Neko 容器的实际网络拓扑，并实现**自动带参免密登录注入**。

## 2. 信令中继工作流设计

```
[React Client]                           [GoStream3D Gateway]                       [Neko Container]
      │                                            │                                        │
      │── 1. WS Connect: /api/v1/ws/:token ───────►│                                        │
      │                                            │── 2. 校验 Token 找到对应会话与内部端口 ──│
      │                                            │── 3. WS Connect (带 pwd 免密参数) ────►│
      │                                            │◄── 4. Connected 确认 ──────────────────│
      │◄── 5. WS Handshake 握手完成 ───────────────│                                        │
      │                                            │                                        │
      │── 6. WebRTC Offer (JSON) ─────────────────►│── (原样双向透传转发) ──────────────────►│
      │◄── 7. WebRTC Answer (JSON) ────────────────│◄── (原样双向透传转发) ─────────────────│
      │◄── 8. ICE Candidates ──────────────────────│◄── (原样双向透传转发) ─────────────────│
      │                                            │                                        │
      │ (心跳 Ping)                                │ (更新 LastActiveAt 时间戳)             │

```

## 3. 信令透明代理实现 (`websocket_proxy.go`)

```go
package proxy

import (
	"fmt"
	"log"
	"net/http"
	"sync"
	"time"

	"GoStream3D/internal/orchestrator"
	"github.com/gin-gonic/gin"
	"github.com/gorilla/websocket"
)

var upgrader = websocket.Upgrader{
	CheckOrigin: func(r *http.Request) bool {
		return true // 允许跨域推流
	},
}

type ProxyHandler struct {
	sessionMgr *orchestrator.SessionManager
}

func NewProxyHandler(sm *orchestrator.SessionManager) *ProxyHandler {
	return &ProxyHandler{sessionMgr: sm}
}

func (h *ProxyHandler) HandleSignalingWS(c *gin.Context) {
	token := c.Param("token")
	session, exists := h.sessionMgr.GetSession(token)
	if !exists {
		c.JSON(http.StatusNotFound, gin.H{"error": "无效或已过期的推流令牌"})
		return
	}

	// 1. 升级客户端 HTTP 到 WebSocket
	clientConn, err := upgrader.Upgrade(c.Writer, c.Request, nil)
	if err != nil {
		log.Printf("升级客户端 WS 失败: %v", err)
		return
	}
	defer clientConn.Close()

	// 2. 连接 Neko 容器内部 WebSocket（注入免密鉴权参数）
	nekoWSURL := fmt.Sprintf("ws://127.0.0.1:%d/api/ws?pwd=user_internal_token", session.HTTPPort)
	nekoConn, _, err := websocket.DefaultDialer.Dial(nekoWSURL, nil)
	if err != nil {
		log.Printf("连接 Neko 内部 WS 失败 (%s): %v", nekoWSURL, err)
		return
	}
	defer nekoConn.Close()

	// 激活会话状态
	h.sessionMgr.KeepAlive(token)

	var closeOnce sync.Once
	closeBoth := func() {
		closeOnce.Do(func() {
			clientConn.Close()
			nekoConn.Close()
			// 标记断开，等待超时垃圾回收
			h.sessionMgr.KeepAlive(token)
		})
	}

	// 双向转发管道
	errChan := make(chan error, 2)

	// Goroutine 1: Client -> Neko (含客户端保活检测)
	go func() {
		for {
			msgType, data, err := clientConn.ReadMessage()
			if err != nil {
				errChan <- err
				return
			}
			// 每次客户端有数据/信令交互，刷新活跃时间
			h.sessionMgr.KeepAlive(token)

			if err := nekoConn.WriteMessage(msgType, data); err != nil {
				errChan <- err
				return
			}
		}
	}()

	// Goroutine 2: Neko -> Client
	go func() {
		for {
			msgType, data, err := nekoConn.ReadMessage()
			if err != nil {
				errChan <- err
				return
			}
			if err := clientConn.WriteMessage(msgType, data); err != nil {
				errChan <- err
				return
			}
		}
	}()

	// 阻塞直到任意一方断开连接
	<-errChan
	closeBoth()
}

```

# 模块三：React 极简纯净视口与交互透传 (`web/src/components`)

## 1. 模块定位

负责呈现纯粹的全屏画面（无外壳），建立基于 WebRTC 的音视频订阅，并复用 Neko 客户端协议直接将鼠标滚轮动作无感注入远端。

## 2. 页面与组件结构

前端无需复杂状态管理，仅需一个沉浸式单组件：`PureStreamPlayer.tsx`。

```tsx
import React, { useEffect, useRef, useState, useCallback } from 'react';

interface PureStreamPlayerProps {
  token: string;
}

export const PureStreamPlayer: React.FC<PureStreamPlayerProps> = ({ token }) => {
  const videoRef = useRef<HTMLVideoElement>(null);
  const pcRef = useRef<RTCPeerConnection | null>(null);
  const wsRef = useRef<WebSocket | null>(null);
  const [isConnected, setIsConnected] = useState<boolean>(false);
  const [errorMsg, setErrorMsg] = useState<string>('');

  // 发送信令包装函数
  const sendSignal = useCallback((event: string, payload: any = {}) => {
    if (wsRef.current && wsRef.current.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify({ event, ...payload }));
    }
  }, []);

  useEffect(() => {
    // 1. 初始化标准 RTCPeerConnection
    const pc = new RTCPeerConnection({
      iceServers: [{ urls: 'stun:stun.l.google.com:19302' }],
    });
    pcRef.current = pc;

    // 绑定远端流至 Video 标签
    pc.ontrack = (event) => {
      if (videoRef.current && event.streams[0]) {
        videoRef.current.srcObject = event.streams[0];
        setIsConnected(true);
      }
    };

    // 交换本地 ICE 候选
    pc.onicecandidate = (event) => {
      if (event.candidate) {
        sendSignal('signal/candidate', { data: JSON.stringify(event.candidate) });
      }
    };

    // 2. 连接 Go 网关信令代理
    const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:';
    const ws = new WebSocket(`${protocol}//${window.location.host}/api/v1/ws/${token}`);
    wsRef.current = ws;

    ws.onmessage = async (e) => {
      try {
        const msg = JSON.parse(e.data);

        // 处理 Neko 服务端发出的 Offer
        if (msg.event === 'signal/offer') {
          await pc.setRemoteDescription(new RTCSessionDescription({ type: 'offer', sdp: msg.sdp }));
          const answer = await pc.createAnswer();
          await pc.setLocalDescription(answer);
          sendSignal('signal/answer', { sdp: answer.sdp });
        } 
        // 处理远端 ICE Candidate
        else if (msg.event === 'signal/candidate') {
          const candidateData = typeof msg.data === 'string' ? JSON.parse(msg.data) : msg.data;
          await pc.addIceCandidate(new RTCIceCandidate(candidateData));
        }
      } catch (err: any) {
        console.error('信令解析异常:', err);
      }
    };

    ws.onerror = () => setErrorMsg('信令服务器连接失败');
    ws.onclose = () => setIsConnected(false);

    // 3. 维持心跳保活
    const heartbeatTimer = setInterval(() => {
      sendSignal('heartbeat', { timestamp: Date.now() });
    }, 15000);

    return () => {
      clearInterval(heartbeatTimer);
      pc.close();
      ws.close();
    };
  }, [token, sendSignal]);

  // 4. 坐标归一化与交互事件透传 (直接走 Neko 的 WebSocket 控制协议)
  const sendMouseEvent = (type: string, e: React.MouseEvent<HTMLVideoElement>, extra: any = {}) => {
    if (!videoRef.current) return;
    const rect = videoRef.current.getBoundingClientRect();
    
    // 计算 1920x1080 虚拟屏幕下的绝对坐标
    const x = Math.round(((e.clientX - rect.left) / rect.width) * 1920);
    const y = Math.round(((e.clientY - rect.top) / rect.height) * 1080);

    sendSignal(type, { x, y, ...extra });
  };

  return (
    <div style={{ width: '100vw', height: '100vh', backgroundColor: '#000', overflow: 'hidden' }}>
      {/* 加载与异常状态遮罩 */}
      {!isConnected && (
        <div style={{
          position: 'absolute', top: '50%', left: '50%', transform: 'translate(-50%, -50%)',
          color: '#fff', fontFamily: 'monospace', zIndex: 10
        }}>
          {errorMsg ? errorMsg : '3D 场景预渲染加载中...'}
        </div>
      )}

      {/* 核心全屏画布 */}
      <video
        ref={videoRef}
        autoPlay
        playsInline
        muted
        onMouseMove={(e) => sendMouseEvent('mouse/move', e)}
        onMouseDown={(e) => sendMouseEvent('mouse/down', e, { key: e.button + 1 })}
        onMouseUp={(e) => sendMouseEvent('mouse/up', e, { key: e.button + 1 })}
        onWheel={(e) => {
          sendMouseEvent('mouse/scroll', e, { 
            delta_x: 0, 
            delta_y: Math.sign(e.deltaY) * 120 
          });
        }}
        onContextMenu={(e) => e.preventDefault()} // 拦截右键菜单，防止干扰 3D 旋转
        style={{
          width: '100vw',
          height: '100vh',
          objectFit: 'fill',
          display: 'block',
          cursor: 'grab',
        }}
      />
    </div>
  );
};

```

## 4. 落地联调建议

1. **先测网关转发通道**：先手动起一个带有 `NEKO_IMPLICIT_CONTROL=true` 的 Neko 容器，将 `HTTPPort` 指向它，优先跑通 `proxy` 的 WebSocket 转发与 React 的画面全屏渲染。
2. **再接 Docker SDK 编排**：把手动起容器的步骤换成 `SessionManager.LaunchNekoContainer`，确认短链生成到容器秒级拉起、自动分配端口流程完全闭环。
